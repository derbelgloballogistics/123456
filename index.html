🛠️ Admin: Supabase setup SQL for PDF sharing (run again after this update)

Run this in Supabase → SQL Editor. It is safe to run more than once.

-- ===== PDF SHARING — run once (safe to run again) ===== -- 1. PDFs sent to receivers (rows are kept as history; the file itself is wiped after 72h) create table if not exists public.pdf_shares ( id text primary key, recipient text not null, sender text not null, file_name text not null, file_size bigint, file_data text, -- PDF as base64, emptied after 72h created_at timestamptz not null default now(), expires_at timestamptz not null, downloaded_at timestamptz ); alter table public.pdf_shares alter column file_data drop not null; alter table public.pdf_shares add column if not exists download_count integer not null default 0; alter table public.pdf_shares add column if not exists last_downloaded_at timestamptz; alter table public.pdf_shares add column if not exists barcodes text[] not null default '{}'; alter table public.pdf_shares add column if not exists duplicate_barcodes text[] not null default '{}'; create index if not exists pdf_shares_recipient_idx on public.pdf_shares (recipient); create index if not exists pdf_shares_expires_idx on public.pdf_shares (expires_at); create index if not exists pdf_shares_barcodes_idx on public.pdf_shares using gin (barcodes); -- 2. Duplicate-barcode warnings shown to everyone until an admin resolves them create table if not exists public.pdf_alerts ( id text primary key, barcode text not null, share_id text, file_name text, sender text, recipient text, previous_share_id text, previous_file_name text, previous_sender text, previous_recipient text, previous_sent_at timestamptz, note text, created_at timestamptz not null default now(), resolved boolean not null default false, resolved_by text, resolved_at timestamptz ); -- 3. Let the app read/write both tables alter table public.pdf_shares enable row level security; alter table public.pdf_alerts enable row level security; drop policy if exists "app access pdf_shares" on public.pdf_shares; create policy "app access pdf_shares" on public.pdf_shares for all to anon, authenticated using (true) with check (true); drop policy if exists "app access pdf_alerts" on public.pdf_alerts; create policy "app access pdf_alerts" on public.pdf_alerts for all to anon, authenticated using (true) with check (true); -- 4. Old "inbox" accounts become "receiver" accounts update public.users set role = 'receiver' where role = 'inbox'; -- 5. Server-side: wipe file contents after 72h, every hour (keeps history). -- Needs the pg_cron extension (Database -> Extensions -> pg_cron). create extension if not exists pg_cron; select cron.schedule('delete-expired-pdfs', '0 * * * *', $$update public.pdf_shares set file_data = null where expires_at < now() and file_data is not null$$); -- Delete requests on parcels (user deletes → admin confirms or restores) alter table public.parcels add column if not exists delete_requested_by text; alter table public.parcels add column if not exists delete_requested_at timestamptz; -- Session persistence: remember which tab each user was on alter table public.users add column if not exists last_session jsonb;
